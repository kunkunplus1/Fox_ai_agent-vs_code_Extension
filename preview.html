好的，下面是一个零依赖的 debounce 实现，支持 leading（立即执行）和 trailing（尾部执行，默认）两种模式：

javascript
export function debounce(fn, wait = 200, options = {}) {
  const { leading = false, trailing = true } = options;
  let timer = null, lastArgs = null;

  function invoke(context, args) {
    fn.apply(context, args);
    lastArgs = null;
  }

  function debounced(...args) {
    const callNow = leading && !timer;
    if (timer) clearTimeout(timer);
    lastArgs = args;
    timer = setTimeout(() => {
      timer = null;
      if (trailing && lastArgs) invoke(this, lastArgs);
    }, wait);
    if (callNow) invoke(this, args);
  }

  debounced.cancel = () => { if (timer) clearTimeout(timer); timer = null; };
  return debounced;
}

用法：const onResize = debounce(handler, 300, { leading: true })。
